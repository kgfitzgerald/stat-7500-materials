Reading in Character Strings from Files

read_file()    # reads entire file as single string
read_lines()   # creates vector of strings, each line separate

Changing the Case

str_to_lower(x)
str_to_upper(x)
str_to_title(x)
str_to_sentence(x)

# Examples:
str_to_lower("Villanova")
str_to_upper(state.name)

![Image placeholder]

Substrings

str_sub(string, start, end)  # negative numbers count from end
str_sub("Villanova", 1, 1)   # extracts first value
str_sub("Villanova", 6, 9)   # extracts nickname
str_sub("Villanova", 6)      # from 6 to end
str_sub("Villanova", -4)     # 4th from end
str_sub(state.name, 1, 3)

# Another option:
abbreviate(state.name)

![Image placeholder]

Find Character Strings

str_which(vector, pattern)      # indices of matches
str_subset(vector, pattern)     # values of matches
str_detect(vector, pattern)     # logical vector

# Examples:
str_which(c("Aardvark","Anteater","Alligator"),"k")  # => 1
str_subset(c("Aardvark","Anteater","Alligator"),"k") # => "Aardvark"
str_detect(c("Aardvark","Anteater","Alligator"),"k") # => T F F

Find Character Strings II

str_which(state.name,"V")
str_subset(state.name,"V")
str_subset(str_to_upper(state.name),"V")
str_subset(state.name,"[Vv]")

Find and Replace

str_replace(string, pattern, replacement)      # first occurrence
str_replace_all(string, pattern, replacement)  # all occurrences

# Remove spaces:
states_nospace <- str_replace_all(state.name, " ", "")
# Or better:
str_remove_all(state.name, " ")

![Image placeholder]

Counting String Occurrence

str_count(string, pattern)
str_count("pneumonoultramicroscopicsilicovolcanoconiosis","[aeiou]")

fruit <- c("apple","banana","pear","pineapple")
str_count(fruit,"a")

has_ss <- str_count(str_to_lower(state.name),"ss")
state.name[has_ss > 0]
state.name[has_ss > 1]

str_count(read_file("https://en.wikipedia.org/wiki/Villanova_University"),"Villanova")
# The word “Villanova” appears 613 times!

Anchor Sequences

str_replace_all("the dandelion war of 2010", "\d", "_")
# => "the dandelion war of ____"

myword <- "t e s t i f y"
str_replace_all(myword,"\w","_")  # hangman
str_replace_all(myword,"[^t\s]","_")
str_replace_all(myword,"[^te\s]","_")

Quantifiers

str_subset(state.name,"e")
str_subset(state.name,"e{2,}")  # Looks for 2 e’s in a row
